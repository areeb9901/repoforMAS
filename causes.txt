@Override
public SentMail getSavedEmailWithMaIdAndSentTime(Long maId, Date sentDateTime) {
    return (SentMail) getSession()
        .createQuery("from SentMail where maOrGrpId = :ids and sentTime = :sentTime and (mailTrigger = :termTrigger or mailTrigger = :dormTrigger)")
        .setLong("ids", maId)
        .setTimestamp("sentTime", sentDateTime)
        .setString("termTrigger", "MA Terminated")
        .setString("dormTrigger", "MA Dormanted")
        .uniqueResult();
}
