CREATE OR REPLACE PROCEDURE SORT_AMENDMENTS_GROUP
(
    grpId NUMBER,
    Cuser VARCHAR2
)
AS
    ---------------------------------------------------------------------------
    -- Mapping between an LDM_TB_AMD row and its final amendment ID
    ---------------------------------------------------------------------------
    TYPE Rec_Amd_Map IS RECORD
    (
        Rowid_Amd  UROWID,
        Old_Amd_Id NUMBER,
        New_Amd_Id NUMBER
    );

    TYPE Typ_Amd_Map IS TABLE OF Rec_Amd_Map
        INDEX BY PLS_INTEGER;

    V_Amd_Map Typ_Amd_Map;

    v_map_index PLS_INTEGER;
    v_temp_base NUMBER;

    ---------------------------------------------------------------------------
    -- Validation temporary-ID variables
    ---------------------------------------------------------------------------
    v_validation_temp_base NUMBER;
    v_validation_index     PLS_INTEGER;

BEGIN
    SAVEPOINT before_sort_amendments_group;

    ---------------------------------------------------------------------------
    -- Clear any reference data remaining in the current transaction
    ---------------------------------------------------------------------------
    DELETE FROM AMD_REF_TABLE;
    DELETE FROM EXEC_DOC_REF_TABLE;
    DELETE FROM AMD_VALDN_REF_TABLE;
    DELETE FROM EXEC_DOC_GRP_REF_TABLE;

    ---------------------------------------------------------------------------
    -- Save group execution-document references before changing amendment IDs.
    --
    -- The amendment ROWID is retained so the final AMD_ID can be retrieved
    -- after sorting.
    ---------------------------------------------------------------------------
    INSERT INTO EXEC_DOC_GRP_REF_TABLE
    (
        ROW_ID,
        EXEC_DATE,
        ROWID_AMD
    )
    SELECT row_id,
           exec_date,
           rowid_amd
    FROM
    (
        SELECT negoGrp.row_id AS row_id,
               amd.amd_d_sign AS exec_date,
               amd.ROWID AS rowid_amd,
               ROW_NUMBER() OVER
               (
                   PARTITION BY negoGrp.row_id
                   ORDER BY amd.date_creat,
                            amd.ROWID
               ) AS rn
        FROM NEGO_TB_ROW_EXEC_DOC_FOR_GRP negoGrp
        JOIN LDM_TB_MA ma
          ON ma.grp_id = negoGrp.grp_id
        JOIN LDM_TB_AMD amd
          ON amd.ma_id = ma.ma_id
         AND amd.amd_id = negoGrp.amd_id
        WHERE negoGrp.grp_id = grpId

          -- Terminated MAs are not sorted.
          AND NVL(ma.status_id, -1) <> 8

          AND amd.amd_id > 0
          AND amd.amd_d_sign IS NOT NULL
          AND amd.amd_d_term IS NULL
    )
    WHERE rn = 1;

    ---------------------------------------------------------------------------
    -- Process every eligible MA in the group.
    --
    -- STATUS_ID = 8 is treated as Terminated, based on the supplied data.
    ---------------------------------------------------------------------------
    FOR Eachma IN
    (
        SELECT ma_id
        FROM LDM_TB_MA
        WHERE grp_id = grpId
          AND in_nego_flg = 'N'
          AND NVL(status_id, -1) <> 8
        ORDER BY ma_id
    )
    LOOP
        -----------------------------------------------------------------------
        -- Clear reference data from the preceding MA
        -----------------------------------------------------------------------
        DELETE FROM EXEC_DOC_REF_TABLE;
        DELETE FROM AMD_VALDN_REF_TABLE;

        -----------------------------------------------------------------------
        -- Save executed-document references before changing AMD_ID
        -----------------------------------------------------------------------
        INSERT INTO EXEC_DOC_REF_TABLE
        (
            ROW_ID,
            EXEC_DATE,
            ROWID_AMD
        )
        SELECT nego.row_id,
               amd.amd_d_sign,
               amd.ROWID
        FROM NEGO_TB_ROW_EXEC_DOC nego
        JOIN LDM_TB_AMD amd
          ON amd.ma_id = nego.ma_id
         AND amd.amd_id = nego.amd_id
        WHERE amd.ma_id = Eachma.ma_id
          AND amd.amd_id > 0
          AND amd.amd_d_term IS NULL;

        -----------------------------------------------------------------------
        -- Save amendment-validation references before changing AMD_ID
        -----------------------------------------------------------------------
        INSERT INTO AMD_VALDN_REF_TABLE
        (
            AMD_ID,
            ROWID_VALDN,
            ROWID_AMD
        )
        SELECT valdn.amendment_id,
               valdn.ROWID,
               amd.ROWID
        FROM AMENDMENT_VALIDATIONS valdn
        JOIN LDM_TB_AMD amd
          ON amd.ma_id = valdn.agreement_id
         AND amd.amd_id = valdn.amendment_id
        WHERE amd.ma_id = Eachma.ma_id
          AND amd.amd_id > 0
          AND amd.amd_d_term IS NULL
          AND valdn.validator IS NOT NULL;

        -----------------------------------------------------------------------
        -- Build final amendment-ID mapping for the current MA.
        --
        -- Amendments are sorted independently for each MA.
        --
        -- Duplicate dates are ordered deterministically using DATE_CREAT and
        -- ROWID.
        -----------------------------------------------------------------------
        V_Amd_Map.DELETE;
        v_map_index := 0;

        FOR Maprow IN
        (
            SELECT amd.ROWID AS rowid_amd,
                   amd.amd_id AS old_amd_id,
                   ROW_NUMBER() OVER
                   (
                       ORDER BY amd.amd_d_sign,
                                amd.date_creat,
                                amd.ROWID
                   ) AS new_amd_id
            FROM LDM_TB_AMD amd
            WHERE amd.ma_id = Eachma.ma_id
              AND amd.amd_id > 0
              AND amd.amd_d_sign IS NOT NULL
              AND amd.amd_d_term IS NULL
            ORDER BY amd.amd_d_sign,
                     amd.date_creat,
                     amd.ROWID
        )
        LOOP
            v_map_index := v_map_index + 1;

            V_Amd_Map(v_map_index).Rowid_Amd :=
                Maprow.rowid_amd;

            V_Amd_Map(v_map_index).Old_Amd_Id :=
                Maprow.old_amd_id;

            V_Amd_Map(v_map_index).New_Amd_Id :=
                Maprow.new_amd_id;
        END LOOP;

        -----------------------------------------------------------------------
        -- Two-phase amendment-ID update.
        --
        -- Phase 1 assigns temporary IDs above the existing maximum. This
        -- prevents ORA-00001 when amendment IDs need to be swapped.
        -----------------------------------------------------------------------
        IF V_Amd_Map.COUNT > 0 THEN

            SELECT NVL(MAX(ABS(amd_id)), 0) + 1000
            INTO v_temp_base
            FROM LDM_TB_AMD
            WHERE ma_id = Eachma.ma_id;

            FOR i IN V_Amd_Map.FIRST .. V_Amd_Map.LAST
            LOOP
                UPDATE LDM_TB_AMD
                SET amd_id     = v_temp_base + i,
                    date_modif = SYSDATE,
                    user_modif = Cuser
                WHERE ROWID = V_Amd_Map(i).Rowid_Amd;
            END LOOP;

            -------------------------------------------------------------------
            -- Phase 2 assigns the final chronologically sorted IDs
            -------------------------------------------------------------------
            FOR i IN V_Amd_Map.FIRST .. V_Amd_Map.LAST
            LOOP
                UPDATE LDM_TB_AMD
                SET amd_id     = V_Amd_Map(i).New_Amd_Id,
                    date_modif = SYSDATE,
                    user_modif = Cuser
                WHERE ROWID = V_Amd_Map(i).Rowid_Amd;
            END LOOP;

        END IF;

        -----------------------------------------------------------------------
        -- Update individual executed-document references using amendment ROWID
        -----------------------------------------------------------------------
        MERGE INTO NEGO_TB_ROW_EXEC_DOC exec_doc
        USING
        (
            SELECT reft.row_id AS ref_row_id,
                   amd.amd_id AS new_amd_id
            FROM EXEC_DOC_REF_TABLE reft
            JOIN LDM_TB_AMD amd
              ON amd.ROWID = reft.rowid_amd
            WHERE amd.ma_id = Eachma.ma_id
        ) data_t
        ON
        (
            exec_doc.row_id = data_t.ref_row_id
            AND exec_doc.ma_id = Eachma.ma_id
        )
        WHEN MATCHED THEN
            UPDATE SET
                exec_doc.amd_id     = data_t.new_amd_id,
                exec_doc.user_modif = Cuser,
                exec_doc.date_modif = SYSDATE;

        -----------------------------------------------------------------------
        -- Two-phase update of Amendment_Validations.
        --
        -- The validation rows are first assigned temporary IDs and are then
        -- assigned the final AMD_ID from the corresponding amendment ROWID.
        -----------------------------------------------------------------------
        SELECT NVL(MAX(ABS(amendment_id)), 0) + 1000
        INTO v_validation_temp_base
        FROM AMENDMENT_VALIDATIONS
        WHERE agreement_id = Eachma.ma_id;

        v_validation_index := 0;

        FOR ValidationRef IN
        (
            SELECT rowid_valdn,
                   rowid_amd
            FROM AMD_VALDN_REF_TABLE
            ORDER BY rowid_valdn
        )
        LOOP
            v_validation_index := v_validation_index + 1;

            UPDATE AMENDMENT_VALIDATIONS
            SET amendment_id =
                    v_validation_temp_base + v_validation_index
            WHERE ROWID = ValidationRef.rowid_valdn;
        END LOOP;

        -----------------------------------------------------------------------
        -- Assign final amendment IDs to validation rows
        -----------------------------------------------------------------------
        FOR ValidationRef IN
        (
            SELECT ref.rowid_valdn,
                   amd.amd_id AS final_amd_id
            FROM AMD_VALDN_REF_TABLE ref
            JOIN LDM_TB_AMD amd
              ON amd.ROWID = ref.rowid_amd
            ORDER BY ref.rowid_valdn
        )
        LOOP
            UPDATE AMENDMENT_VALIDATIONS
            SET amendment_id = ValidationRef.final_amd_id
            WHERE ROWID = ValidationRef.rowid_valdn;
        END LOOP;

        -----------------------------------------------------------------------
        -- Clear current MA reference data
        -----------------------------------------------------------------------
        DELETE FROM EXEC_DOC_REF_TABLE;
        DELETE FROM AMD_VALDN_REF_TABLE;

        V_Amd_Map.DELETE;
    END LOOP;

    ---------------------------------------------------------------------------
    -- Update group executed-document references using the amendment ROWID
    -- retained before sorting.
    ---------------------------------------------------------------------------
    MERGE INTO NEGO_TB_ROW_EXEC_DOC_FOR_GRP exec_doc_grp
    USING
    (
        SELECT ref_row_id,
               new_amd_id
        FROM
        (
            SELECT reft.row_id AS ref_row_id,
                   amd.amd_id AS new_amd_id,
                   ROW_NUMBER() OVER
                   (
                       PARTITION BY reft.row_id
                       ORDER BY amd.amd_id,
                                amd.ROWID
                   ) AS rn
            FROM EXEC_DOC_GRP_REF_TABLE reft
            JOIN LDM_TB_AMD amd
              ON amd.ROWID = reft.rowid_amd
            WHERE amd.amd_d_term IS NULL
        )
        WHERE rn = 1
    ) data_t
    ON
    (
        exec_doc_grp.row_id = data_t.ref_row_id
        AND exec_doc_grp.grp_id = grpId
    )
    WHEN MATCHED THEN
        UPDATE SET
            exec_doc_grp.amd_id     = data_t.new_amd_id,
            exec_doc_grp.user_modif = Cuser,
            exec_doc_grp.date_modif = SYSDATE;

    ---------------------------------------------------------------------------
    -- Final cleanup
    ---------------------------------------------------------------------------
    DELETE FROM AMD_REF_TABLE;
    DELETE FROM EXEC_DOC_REF_TABLE;
    DELETE FROM AMD_VALDN_REF_TABLE;
    DELETE FROM EXEC_DOC_GRP_REF_TABLE;

    COMMIT;

EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK TO before_sort_amendments_group;
        RAISE;
END SORT_AMENDMENTS_GROUP;
